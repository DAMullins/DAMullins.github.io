def calculate_fibonacci(n):
    """Generates the Fibonacci sequence up to n terms."""
    sequence = [0, 1]
    while len(sequence) < n:
        next_value = sequence[-1] + sequence[-2]
        sequence.append(next_value)
    return sequence

print(calculate_fibonacci(10))
